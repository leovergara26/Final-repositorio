name: Construir APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: "8.9"

      - name: Compilar
        run: gradle assembleRelease --no-daemon

      - name: Renombrar APK
        run: cp app/build/outputs/apk/release/app-release.apk caja-papeleria.apk

      - uses: actions/upload-artifact@v4
        with:
          name: caja-papeleria-apk
          path: caja-papeleria.apk

      - name: Publicar en Releases
        uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ github.run_number }}
          name: Caja Papelería v${{ github.run_number }}
          files: caja-papeleria.apk
